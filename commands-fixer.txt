cat > /usr/local/bin/fixpanel <<'EOF'
#!/bin/bash
cd /var/www/pterodactyl || exit 1
export COMPOSER_ALLOW_SUPERUSER=1

# fix the "cannot create cache directory" warning
mkdir -p /var/www/.cache/composer

# packages: remove dev packages, make sure the Discord module exists
composer install --no-dev --optimize-autoloader --no-interaction
composer require socialiteproviders/discord --update-no-dev --no-interaction
composer dump-autoload -o --no-dev --no-interaction

# database and caches
php artisan migrate --force
php artisan optimize:clear
php artisan view:clear

# ownership and permissions (last, so root-created cache files get fixed too)
chown -R www-data:www-data /var/www/pterodactyl/* /var/www/.cache
chmod -R 755 storage/* bootstrap/cache

# restart workers and PHP so nothing serves stale code
systemctl restart pteroq 2>/dev/null
for s in $(systemctl list-units --type=service --no-legend | awk '/php.*fpm/{print $1}'); do
  systemctl restart "$s"
done

echo "---- recent errors (blank = none) ----"
grep -A3 "production.ERROR" storage/logs/laravel-$(date +%F).log 2>/dev/null | tail -n 20
echo "Done."
EOF
chmod +x /usr/local/bin/fixpanel && fixpanel
